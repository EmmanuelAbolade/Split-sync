Generates a PayPal.me deep link for a direct payment.

Generates a PayPal.me deep link for a direct payment. Constructs a PayPal.me URL in the format: https://paypal.me/{username}/{amount}{currency} The @ prefix is stripped from the username if present. Opening this link redirects the payer to the PayPal app or website with the amount pre-filled.

Parameters
{string}username - PayPal.me username (with or without leading @).
{number}amount - Payment amount.
{string}[currency='EUR'] - ISO 4217 currency code.
Returns
{string|null} Full PayPal.me URL, or null if username is not provided.

generatePayPalLink('alice123', 30.00, 'EUR'); // 'https://paypal.me/alice123/30EUR'

// src/utils/paymentLinks.js
export const generatePayPalLink = (username, amount, currency = 'EUR') => {
if (!username) return null;
const cleanUsername = username.replace('@', '');
return `https://paypal.me/${cleanUsername}/${amount}${currency}`;
};
export const generateRevolutLink = (username, amount, currency = 'EUR') => {

const cleanUsername = username.replace('@', '');
return `https://revolut.me/${cleanUsername}`;
};

export const getAvailablePaymentMethods = (paymentMethods) => {
if (!paymentMethods) return [];
const available = [];
if (paymentMethods.paypalUsername) {
available.push({
id: 'paypal',
name: 'PayPal',
username: paymentMethods.paypalUsername,
description: `Pay via PayPal (@${paymentMethods.paypalUsername})`,
});
}
if (paymentMethods.revolutUsername) {
available.push({
id: 'revolut',
name: 'Revolut',
username: paymentMethods.revolutUsername,
description: `Pay via Revolut (@${paymentMethods.revolutUsername})`,
});
}
if (paymentMethods.iban) {
available.push({
id: 'bank',
name: 'Bank Transfer',
iban: paymentMethods.iban,
bic: paymentMethods.bic,
description: 'SEPA bank transfer',
});
}
return available;
};
