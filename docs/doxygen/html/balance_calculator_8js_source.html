18/**
19 * @brief Calculates the net balance for every member of a group.
20 *
21 * @details Iterates over all non-deleted expenses. For each expense the payer is
22 * credited the full amount and each participant is debited their individual split
23 * share. A positive balance means the member is owed money; a negative balance
24 * means the member owes money.
25 *
26 * @param {Array<Object>} expenses - Array of expense documents from Firestore.
27 * Each expense must contain:
28 * - {string} paidBy - UID of the member who paid.
29 * - {number} amount - Total expense amount.
30 * - {Object} splits - Map of { [userId]: shareAmount }.
31 * - {Array} participants - Array of participant UIDs.
32 * - {string} [status] - Expenses with status 'deleted' are excluded.
33 * @param {Array<Object>} members - Array of group member objects, each with a `userId` field.
34 *
35 * @returns {Object} A map of { [userId]: netBalance } where positive = owed, negative = owes.
36 *
37 * @example
38 * const balances = calculateBalances(expenses, group.members);
39 * // { 'uid-alice': 30.00, 'uid-bob': -30.00 }
40 */
41export function calculateBalances(expenses, members) {
42 const balances = {};
43 members.forEach(member => {
44 balances[member.userId] = 0;
45 });
46
47 const activeExpenses = expenses.filter(e => !e.status || e.status !== 'deleted');
48
49 activeExpenses.forEach(expense => {
50 const { paidBy, splits, participants } = expense;
51 balances[paidBy] = (balances[paidBy] || 0) + expense.amount;
52 participants?.forEach(userId => {
53 const share = splits[userId] || 0;
54 balances[userId] = (balances[userId] || 0) - share;
55 });
56 });
57
58 return balances;
59}

61/**
62 * @brief Computes the minimum set of transactions to settle all debts in a group.
63 *
64 * @details Implements a greedy two-pointer algorithm:
65 * 1. Separate members into creditors (positive balance) and debtors (negative balance).
66 * 2. Sort both lists in descending order of amount.
67 * 3. At each step, match the largest creditor with the largest debtor.
68 * 4. Create a settlement for min(creditor.amount, debtor.amount).
69 * 5. Advance the pointer whose balance reaches zero.
70 *
71 * This minimises the total number of transactions to at most n-1,
72 * where n is the number of members with non-zero balances.
73 *
74 * @param {Object} balances - Output of calculateBalances(). Map of { [userId]: netBalance }.
75 *
76 * @returns {Array<Object>} Array of settlement objects, each containing:
77 * - {string} from - UID of the member who must pay.
78 * - {string} to - UID of the member who must receive.
79 * - {number} amount - Amount to transfer, rounded to 2 decimal places.
80 *
81 * @example
82 * const settlements = calculateSettlements(balances);
83 * // [{ from: 'uid-bob', to: 'uid-alice', amount: 30.00 }]
84 */
85export function calculateSettlements(balances) {
86 const creditors = [];
87 const debtors = [];
88
89 Object.entries(balances).forEach(([userId, balance]) => {
90 if (balance > 0.01) creditors.push({ userId, amount: balance });
91 else if (balance < -0.01) debtors.push({ userId, amount: Math.abs(balance) });
92 });
93
94 creditors.sort((a, b) => b.amount - a.amount);
95 debtors.sort((a, b) => b.amount - a.amount);
96
97 const settlements = [];
98 let i = 0;
99 let j = 0;
100
101 while (i < creditors.length && j < debtors.length) {
102 const creditor = creditors[i];
103 const debtor = debtors[j];
104 const amount = Math.min(creditor.amount, debtor.amount);
105
106 settlements.push({
107 from: debtor.userId,
108 to: creditor.userId,
109 amount: parseFloat(amount.toFixed(2)),
110 });
111
112 creditor.amount -= amount;
113 debtor.amount -= amount;
114
115 if (creditor.amount < 0.01) i++;
116 if (debtor.amount < 0.01) j++;
117 }
118
119 return settlements;
120}

122/**
123 * @brief Returns a human-readable balance summary for a specific user.
124 *
125 * @details Classifies the user's net balance into one of three states:
126 * - settled: balance is within +/-0.01 (floating-point safe zero).
127 * - owed: balance is positive; the user is owed money.
128 * - owes: balance is negative; the user owes money.
129 *
130 * @param {Object} balances - Map of { [userId]: netBalance } from calculateBalances().
131 * @param {string} userId - UID of the user whose summary is required.
132 * @param {Array} members - Array of group member objects (reserved for future use).
133 *
134 * @returns {Object} Summary object:
135 * - {string} status - One of 'settled' | 'owed' | 'owes'.
136 * - {string} message - Display string for the UI.
137 * - {number} amount - Absolute balance amount (always non-negative).
138 */
139export function getUserBalanceSummary(balances, userId, members) {
140 const balance = balances[userId] || 0;
141 if (Math.abs(balance) < 0.01) return { status: 'settled', message: 'All settled up!', amount: 0 };
142 if (balance > 0) return { status: 'owed', message: 'You are owed', amount: balance };
143 return { status: 'owes', message: 'You owe', amount: Math.abs(balance) };
144}

146/**
147 * @brief Groups an array of amount/currency pairs by currency code.
148 *
149 * @details Prevents cross-currency summation. EUR and GBP amounts are kept
150 * as separate entries rather than being combined into a single total.
151 * Used across Profile, Activity, and Analytics pages.
152 *
153 * @param {Array<Object>} items - Array of objects each with:
154 * - {number} amount - Monetary value.
155 * - {string} currency - ISO 4217 currency code (e.g. 'EUR', 'GBP', 'USD').
156 *
157 * @returns {Object} Map of { [currencyCode]: totalAmount }.
158 *
159 * @example
160 * groupByCurrency([{ amount: 18.33, currency: 'EUR' }, { amount: 258.33, currency: 'GBP' }]);
161 * // { EUR: 18.33, GBP: 258.33 }
162 */
163export function groupByCurrency(items) {
164 return items.reduce((acc, { amount, currency }) => {
165 if (!currency || !amount) return acc;
166 acc[currency] = (acc[currency] || 0) + amount;
167 return acc;
168 }, {});
169}
