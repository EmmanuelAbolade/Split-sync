41export function calculateBalances(expenses, members) {
42 const balances = {};
43 members.forEach(member => {
44 balances[member.userId] = 0;
45 });
46
47 const activeExpenses = expenses.filter(e => !e.status || e.status !== 'deleted');
48
49 activeExpenses.forEach(expense => {
50 const { paidBy, splits, participants } = expense;
51 balances[paidBy] = (balances[paidBy] || 0) + expense.amount;
52 participants?.forEach(userId => {
53 const share = splits[userId] || 0;
54 balances[userId] = (balances[userId] || 0) - share;
55 });
56 });
57
58 return balances;
59}

85export function calculateSettlements(balances) {
86 const creditors = [];
87 const debtors = [];
88
89 Object.entries(balances).forEach(([userId, balance]) => {
90 if (balance > 0.01) creditors.push({ userId, amount: balance });
91 else if (balance < -0.01) debtors.push({ userId, amount: Math.abs(balance) });
92 });
93
94 creditors.sort((a, b) => b.amount - a.amount);
95 debtors.sort((a, b) => b.amount - a.amount);
96
97 const settlements = [];
98 let i = 0;
99 let j = 0;
100
101 while (i < creditors.length && j < debtors.length) {
102 const creditor = creditors[i];
103 const debtor = debtors[j];
104 const amount = Math.min(creditor.amount, debtor.amount);
105
106 settlements.push({
107 from: debtor.userId,
108 to: creditor.userId,
109 amount: parseFloat(amount.toFixed(2)),
110 });
111
112 creditor.amount -= amount;
113 debtor.amount -= amount;
114
115 if (creditor.amount < 0.01) i++;
116 if (debtor.amount < 0.01) j++;
117 }
118
119 return settlements;
120}
