251 // For 'owed': find settlements where the current user is the payee

254 : settlements.filter(s => s.to === user.uid);

256 relevant.forEach(settlement => {
257 const otherUserId = type === 'owing' ? settlement.to : settlement.from;
258 const member = group.members.find(m => m.userId === otherUserId);
259 if (!member) return;
260
261 let personEntry = breakdowns.find(b => b.userId === otherUserId);
262 if (!personEntry) {
263 personEntry = {
264 userId: otherUserId,
265 name: member.name || 'Unknown',
266 email: member.email || '',
267 photoURL: member.photoURL || null,
268 amountsByCurrency: {}, // { GBP: X, EUR: Y } — replaces single amount
269 groups: [],
270 };
271 breakdowns.push(personEntry);
272 }
273
274 // Accumulate per currency — never sum across currencies
275 const currency = group.currency || 'EUR';
276 personEntry.amountsByCurrency[currency] =
277 (personEntry.amountsByCurrency[currency] || 0) + settlement.amount;
278 personEntry.groups.push({
279 groupId: group.id,
280 groupName: group.name,
281 amount: settlement.amount,
282 currency: group.currency,
283 });

290 const sumB = Object.values(b.amountsByCurrency).reduce((s, v) => s + v, 0);
291 return sumB - sumA;
