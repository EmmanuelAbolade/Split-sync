25// Layout wrapper for protected routes — renders bottom nav, session timeout, and AI chatbot
26function ProtectedLayout({ children }) {
27 const location = useLocation();
28 const { user } = useAuthStore();
29 const { subscribeToNotifications } = useNotificationsStore();
30
31 // Subscribe to notifications at the app level — not just on Dashboard.
32 // Ensures payment_log and other notifications update in real time
33 // on any page, without needing a refresh.
34 useEffect(() => {
35 if (!user?.uid) return;
36 const unsubscribe = subscribeToNotifications(user.uid);
37 return () => unsubscribe();
38 }, [user?.uid, subscribeToNotifications]);
39
40 // Session timeout hook — drives the auto-logout warning modal
41 const {
42 showWarning,
43 countdown,
44 stayLoggedIn,
45 logoutNow,
46 } = useSessionTimeout();
47
48 // Hide bottom nav on group details and invite pages
49 const hideBottomNav = location.pathname.startsWith('/group/') ||
50 location.pathname.startsWith('/invite/');
51
52 return (
53 <>
54 {children}
55 {!hideBottomNav && <BottomNav />}
56 <SplitSyncChat />
57 {/* Session timeout warning modal */}
58 {showWarning && (
59 <SessionTimeoutWarning
60 countdown={countdown}
61 onStayLoggedIn={stayLoggedIn}
62 onLogout={logoutNow}
63 />
64 )}
65 </>
66 );
67}
