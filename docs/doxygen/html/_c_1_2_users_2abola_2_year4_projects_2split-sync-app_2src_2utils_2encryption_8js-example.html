Encrypts a plaintext string using AES-256 before Firestore storage.

Encrypts a plaintext string using AES-256 before Firestore storage. Uses CryptoJS.AES.encrypt with the application encryption key. Empty or whitespace-only input returns an empty string without encrypting. This function is called before any IBAN value is written to Firestore.

Parameters
{string}data - Plaintext data to encrypt (e.g. a raw IBAN string).
Returns
{string} AES-encrypted ciphertext as a Base64 string, or '' if input is empty.
Exceptions
{Error}Throws 'Failed to encrypt data' if CryptoJS encounters an error.

const encrypted = encryptData('IE29AIBK93115212345678'); // 'U2FsdGVkX1+...' (Base64 ciphertext)

// src/utils/encryption.js
import CryptoJS from 'crypto-js';
const ENCRYPTION_KEY = process.env.REACT_APP_ENCRYPTION_KEY || 'split-sync-secure-key-2026';
export const encryptData = (data) => {
if (!data || data.trim() === '') return '';
try {
return CryptoJS.AES.encrypt(data.trim(), ENCRYPTION_KEY).toString();
} catch (error) {
console.error('Encryption error:', error);
throw new Error('Failed to encrypt data');
}
};
export const decryptData = (encryptedData) => {
if (!encryptedData || encryptedData.trim() === '') return '';
try {
const bytes = CryptoJS.AES.decrypt(encryptedData, ENCRYPTION_KEY);
return bytes.toString(CryptoJS.enc.Utf8);
} catch (error) {
console.error('Decryption error:', error);
throw new Error('Failed to decrypt data');
}
};
export const validateIBAN = (iban) => {
if (!iban) return true;
const cleanIBAN = iban.replace(/\s/g, '').toUpperCase();
const ibanRegex = /^[A-Z]{2}[0-9]{2}[A-Z0-9]{1,30}$/;
return ibanRegex.test(cleanIBAN);
};
export const formatIBANDisplay = (iban) => {
if (!iban) return '';
const cleanIBAN = iban.replace(/\s/g, '').toUpperCase();
return cleanIBAN.match(/.{1,4}/g)?.join(' ') || cleanIBAN;
};
export const maskIBAN = (iban) => {
if (!iban) return '';
const cleanIBAN = iban.replace(/\s/g, '');
if (cleanIBAN.length <= 4) return cleanIBAN;
const countryCode = cleanIBAN.substring(0, 2);
const lastFour = cleanIBAN.substring(cleanIBAN.length - 4);
const middleMask = '*'.repeat(Math.min(cleanIBAN.length - 6, 12));
return `${countryCode}${middleMask}${lastFour}`;
};
