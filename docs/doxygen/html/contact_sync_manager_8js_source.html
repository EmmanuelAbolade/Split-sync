10export const hashPhoneNumber = async (phoneNumber) => {
11 try {
12 // Remove all spaces, dashes, and special characters
13 const cleanPhone = phoneNumber.replace(/[\s\-()]/g, '');
14 // Ensure it starts with +
15 const normalizedPhone = cleanPhone.startsWith('+') ? cleanPhone : `+${cleanPhone}`;
16
17 // Hash using SHA-256
18 const encoder = new TextEncoder();
19 const data = encoder.encode(normalizedPhone);
20 const hashBuffer = await crypto.subtle.digest('SHA-256', data);
21 const hashArray = Array.from(new Uint8Array(hashBuffer));
22 const phoneHash = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
23
24 return phoneHash;
25 } catch (error) {
26 console.error('Error hashing phone number:', error);
27 throw error;
28 }
