22// src/utils/encryption.js
23import CryptoJS from 'crypto-js';
24
25const ENCRYPTION_KEY = process.env.REACT_APP_ENCRYPTION_KEY || 'split-sync-secure-key-2026';

27/**
28 * @brief Encrypts a plaintext string using AES-256 before Firestore storage.
29 *
30 * @details Uses CryptoJS.AES.encrypt with the application encryption key.
31 * Empty or whitespace-only input returns an empty string without encrypting.
32 * This function is called before any IBAN value is written to Firestore.
33 *
34 * @param {string} data - Plaintext data to encrypt (e.g. a raw IBAN string).
35 *
36 * @returns {string} AES-encrypted ciphertext as a Base64 string, or '' if input is empty.
37 *
38 * @throws {Error} Throws 'Failed to encrypt data' if CryptoJS encounters an error.
39 *
40 * @example
41 * const encrypted = encryptData('IE29AIBK93115212345678');
42 * // 'U2FsdGVkX1+...' (Base64 ciphertext)
43 */
44export const encryptData = (data) => {
45 if (!data || data.trim() === '') return '';
46 try {
47 return CryptoJS.AES.encrypt(data.trim(), ENCRYPTION_KEY).toString();
48 } catch (error) {
49 console.error('Encryption error:', error);
50 throw new Error('Failed to encrypt data');
51 }
52};
53
54/**
55 * @brief Decrypts an AES-256 ciphertext string retrieved from Firestore.
56 *
57 * @details Reverses the operation performed by encryptData(). Called when
58 * displaying payment settings to the authenticated user. Empty or
59 * whitespace-only input returns an empty string without attempting decryption.
60 *
61 * @param {string} encryptedData - AES ciphertext (Base64 string from Firestore).
62 *
63 * @returns {string} Plaintext string, or '' if input is empty.
64 *
65 * @throws {Error} Throws 'Failed to decrypt data' if decryption fails
66 * (e.g. wrong key or corrupted ciphertext).
67 *
68 * @example
69 * const iban = decryptData('U2FsdGVkX1+...');
70 * // 'IE29AIBK93115212345678'
71 */
72export const decryptData = (encryptedData) => {
73 if (!encryptedData || encryptedData.trim() === '') return '';
74 try {
75 const bytes = CryptoJS.AES.decrypt(encryptedData, ENCRYPTION_KEY);
76 return bytes.toString(CryptoJS.enc.Utf8);
77 } catch (error) {
78 console.error('Decryption error:', error);
79 throw new Error('Failed to decrypt data');
80 }
81};

83/**
84 * @brief Validates whether a string conforms to the basic IBAN format.
85 *
86 * @details Performs a structural check only (not a full modulo-97 checksum
87 * verification). The IBAN is normalised to uppercase and stripped of spaces
88 * before validation. An empty value returns true (field is optional).
89 *
90 * Supported format: 2 uppercase letters (country code) + 2 digits (check digits)
91 * + 1 to 30 alphanumeric characters (BBAN).
92 *
93 * @param {string} iban - IBAN string to validate (may contain spaces).
94 *
95 * @returns {boolean} True if the format is valid or the input is empty; false otherwise.
96 *
97 * @example
98 * validateIBAN('IE29 AIBK 9311 5212 3456 78'); // true
99 * validateIBAN('INVALID'); // false
100 * validateIBAN(''); // true (empty is allowed)
101 */
102export const validateIBAN = (iban) => {
103 if (!iban) return true;
104 const cleanIBAN = iban.replace(/\s/g, '').toUpperCase();
105 const ibanRegex = /^[A-Z]{2}[0-9]{2}[A-Z0-9]{1,30}$/;
106 return ibanRegex.test(cleanIBAN);
107};

109/**
110 * @brief Formats an IBAN with spaces every four characters for human-readable display.
111 *
112 * @param {string} iban - Raw IBAN string (may already contain spaces).
113 *
114 * @returns {string} IBAN grouped in blocks of four, e.g. 'IE29 AIBK 9311 5212 3456 78'.
115 *
116 * @example
117 * formatIBANDisplay('IE29AIBK93115212345678');
118 * // 'IE29 AIBK 9311 5212 3456 78'
119 */
120export const formatIBANDisplay = (iban) => {
121 if (!iban) return '';
122 const cleanIBAN = iban.replace(/\s/g, '').toUpperCase();
123 return cleanIBAN.match(/.{1,4}/g)?.join(' ') || cleanIBAN;
124};

126/**
127 * @brief Masks an IBAN for secure display, showing only the country code and last four digits.
128 *
129 * @details Used in the Payment Settings UI to confirm the IBAN is saved without
130 * exposing the full account number. The middle section is replaced with asterisks.
131 *
132 * @param {string} iban - Raw IBAN string.
133 *
134 * @returns {string} Masked IBAN, e.g. 'IE**3456' for a short IBAN.
135 *
136 * @example
137 * maskIBAN('IE29AIBK93115212345678');
138 * // 'IE************5678'
139 */
140export const maskIBAN = (iban) => {
141 if (!iban) return '';
142 const cleanIBAN = iban.replace(/\s/g, '');
143 if (cleanIBAN.length <= 4) return cleanIBAN;
144 const countryCode = cleanIBAN.substring(0, 2);
145 const lastFour = cleanIBAN.substring(cleanIBAN.length - 4);
146 const middleMask = '*'.repeat(Math.min(cleanIBAN.length - 6, 12));
147 return `${countryCode}${middleMask}${lastFour}`;
148};
