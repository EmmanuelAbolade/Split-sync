8export default function ProfilePhotoUpload({ user, currentPhotoURL, onPhotoUpdated }) {

10 const [error, setError] = useState('');
11 const fileInputRef = useRef(null);
12
13 const handlePhotoSelect = async (e) => {
14 const file = e.target.files[0];
15 if (!file) return;
16
17 // Validate file type
18 if (!file.type.startsWith('image/')) {
19 setError('Please select an image file');
20 return;
21 }
22
23 // Validate file size (max 5MB)
24 if (file.size > 5 * 1024 * 1024) {
25 setError('Image size must be less than 5MB');
26 return;
27 }
28
29 setError('');
30 setUploading(true);
31
32 try {
33 // Create storage reference
34 const storageRef = ref(storage, `profilePhotos/${user.uid}`);
35
36 // Upload file
37 await uploadBytes(storageRef, file);
38
39 // Get download URL
40 const photoURL = await getDownloadURL(storageRef);
41
42 // Update Firestore user document
43 await updateDoc(doc(db, 'users', user.uid), {
44 photoURL: photoURL,
45 updatedAt: new Date(),
46 });
47
48 // Callback to parent
49 if (onPhotoUpdated) {
50 onPhotoUpdated(photoURL);
51 }
52
53 setUploading(false);
54 } catch (err) {
55 console.error('Upload error:', err);
56 setError('Failed to upload photo. Please try again.');
57 setUploading(false);
58 }
59 };
60
61 return (
62 <div className="relative">
63 <input
64 ref={fileInputRef}
65 type="file"
66 accept="image/*"
67 onChange={handlePhotoSelect}
68 className="hidden"
69 />
70
71 <div className="relative inline-block">

73 <div className="w-24 h-24 rounded-full overflow-hidden bg-gradient-to-br from-primary to-primary-dark dark:from-indigo-600 dark:to-indigo-800 shadow-card-hover dark:shadow-gray-900/50">
74 {currentPhotoURL ? (
75 <img
76 src={currentPhotoURL}
77 alt="Profile"
78 className="w-full h-full object-cover"
79 />
80 ) : (
81 <div className="w-full h-full flex items-center justify-center">
82 <span className="text-4xl font-bold text-white">
83 {user?.displayName?.charAt(0).toUpperCase() || user?.email?.charAt(0).toUpperCase()}
84 </span>
85 </div>
86 )}
87 </div>
88
89 {/* Upload Button Overlay */}
90 <button
91 onClick={() => fileInputRef.current?.click()}
92 disabled={uploading}

104 <p className="text-red-600 dark:text-red-400 text-xs mt-2">{error}</p>
