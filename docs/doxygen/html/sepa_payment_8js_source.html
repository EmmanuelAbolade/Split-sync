57export const maskIBAN = (iban) => {
58 if (!iban) return '';
59
60 const cleanIBAN = iban.replace(/\s/g, '');
61
62 if (cleanIBAN.length <= 6) return cleanIBAN;
63
64 const countryCode = cleanIBAN.substring(0, 2);
65 const lastFour = cleanIBAN.substring(cleanIBAN.length - 4);
66 const middleLength = cleanIBAN.length - 6;
67 const mask = '*'.repeat(Math.min(middleLength, 12));
68
69 const masked = `${countryCode}${mask}${lastFour}`;
70 return formatIBAN(masked);
71};
