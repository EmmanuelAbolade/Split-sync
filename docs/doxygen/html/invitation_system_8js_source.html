23export const sendInvitation = async (fromUser, toUserId, groupId, groupName) => {
24
25 try {
26 // Check if user is already in the group
27 const groupDoc = await getDoc(doc(db, 'groups', groupId));
28 if (groupDoc.exists()) {
29 const groupData = groupDoc.data();
30 if (groupData.memberIds?.includes(toUserId)) {
31 return { success: false, message: 'User is already in this group' };
32 }
33 }
34
35 // Get recipient info
36 const toUserDoc = await getDoc(doc(db, 'users', toUserId));
37 if (!toUserDoc.exists()) {
38 return { success: false, message: 'User not found' };
39 }
40 const toUserData = toUserDoc.data();
41
42
43
44 // Create invitation
45 const invitationRef = doc(collection(db, 'invitations'));
46 const invitationData = {
47 id: invitationRef.id,
48 groupId,
49 groupName,
50 fromUserId: fromUser.uid,
51 fromUserName: fromUser.displayName || fromUser.email,
52 fromUserPhoto: fromUser.photoURL || null,
53 toUserId,
54 toUserEmail: toUserData.email,
55 toUserName: toUserData.fullName || toUserData.displayName || 'User',
56 status: 'pending',
57 createdAt: serverTimestamp(),
58 expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000), // 7 days
59 notificationSent: false
60 };
61
62 try {
63 await setDoc(invitationRef, invitationData);
64 } catch (inviteError) {
65 throw inviteError; // Re-throw to maintain existing error handling
66 }
67
68 // Create bell notification for recipient
69 try {
70 console.log('🔔 [INVITE] Notification data:', {
71 userId: toUserId,
72 type: 'invitation',
73 groupName: groupName
74 });
75
76 await createNotification({
77 userId: toUserId,
78 type: 'invitation',
79 title: '📨 Group Invitation',
80 message: `${fromUser.displayName || 'Someone'} invited you to join "${groupName}"`,
81 data: {
82 invitationId: invitationRef.id,
83 groupId: groupId,
84 groupName: groupName,
85 fromUserId: fromUser.uid,
86 fromUserName: fromUser.displayName || fromUser.email,
87 },
88 });
89
90} catch (notifError) {
91 // Don't fail the invitation if notification fails
92}
93
94console.log(`✅ Invitation sent: ${fromUser.displayName} → ${toUserData.fullName} for ${groupName}`);
95
96return {
97 success: true,
98 invitationId: invitationRef.id,
99 message: 'Invitation sent successfully'
100};
101
102 } catch (error) {
103 console.error('Error sending invitation:', error);
104 return { success: false, message: 'Failed to send invitation' };
105 }

111export const sendBulkInvitations = async (fromUser, toUserIds, groupId, groupName) => {
112 const results = {
113 successful: [],
114 failed: []
115 };
116
117 for (const toUserId of toUserIds) {
118 const result = await sendInvitation(fromUser, toUserId, groupId, groupName);
119
120 if (result.success) {
121 results.successful.push(toUserId);
122 } else {
123 results.failed.push({ userId: toUserId, reason: result.message });
124 }
125 }
126
127 return results;

133export const getMyInvitations = async (userId) => {
134 try {
135 const invitationsQuery = query(
136 collection(db, 'invitations'),
137 where('toUserId', '==', userId),
138 where('status', '==', 'pending')
139 );
140
141 const snapshot = await getDocs(invitationsQuery);
142 const invitations = [];
143
144 for (const docSnap of snapshot.docs) {
145 const data = docSnap.data();
146
147 // Check if expired
148 const expiresAt = data.expiresAt?.toMillis() || 0;
149 if (expiresAt < Date.now()) {
150 // Mark as expired
151 await updateDoc(doc(db, 'invitations', docSnap.id), {
152 status: 'expired'
153 });
154 continue;
155 }
156
157 invitations.push({
158 id: docSnap.id,
159 ...data
160 });
161 }
162
163 // Sort by most recent
164 invitations.sort((a, b) => {
165 const aTime = a.createdAt?.toMillis() || 0;
166 const bTime = b.createdAt?.toMillis() || 0;
167 return bTime - aTime;
168 });
169
170 return invitations;
171 } catch (error) {
172 console.error('Error getting invitations:', error);
173 return [];
174 }

210export const acceptInvitation = async (invitationId, userId) => {
211 try {
212 // Get invitation
213 const invitationDoc = await getDoc(doc(db, 'invitations', invitationId));
214 if (!invitationDoc.exists()) {
215 return { success: false, message: 'Invitation not found' };
216 }
217
218 const invitationData = invitationDoc.data();
219
220 // Verify it's for this user
221 if (invitationData.toUserId !== userId) {
222 return { success: false, message: 'This invitation is not for you' };
223 }
224
225 // Check if expired
226 const expiresAt = invitationData.expiresAt?.toMillis() || 0;
227 if (expiresAt < Date.now()) {
228 await updateDoc(doc(db, 'invitations', invitationId), { status: 'expired' });
229 return { success: false, message: 'Invitation has expired' };
230 }
231
232 // Get group
233 const groupRef = doc(db, 'groups', invitationData.groupId);
234 const groupDoc = await getDoc(groupRef);
235
236 if (!groupDoc.exists()) {
237 return { success: false, message: 'Group not found' };
238 }
239
240 const groupData = groupDoc.data();
241
242 // Check if already a member
243 if (groupData.memberIds?.includes(userId)) {
244 await updateDoc(doc(db, 'invitations', invitationId), { status: 'accepted' });
245 return { success: false, message: 'You are already a member of this group' };
246 }
247
248 // Get user info
249 const userDoc = await getDoc(doc(db, 'users', userId));
250 const userData = userDoc.data();
251
252 // Add user to group
253 await updateDoc(groupRef, {
254 memberIds: arrayUnion(userId),
255 members: arrayUnion({
256 userId,
257 name: userData.displayName || userData.fullName || 'User',
258 email: userData.email,
259 photoURL: userData.photoURL || null,
260 joinedAt: new Date()
261 }),
262 updatedAt: serverTimestamp()
263 });
264
265 // Mark invitation as accepted
266 await updateDoc(doc(db, 'invitations', invitationId), {
267 status: 'accepted',
268 acceptedAt: serverTimestamp()
269 });
270
271 console.log(`✅ Invitation accepted: ${userData.fullName} joined ${groupData.name}`);
272
273 return {
274 success: true,
275 groupId: invitationData.groupId,
276 groupName: invitationData.groupName,
277 message: `You've joined ${invitationData.groupName}!`
278 };
279 } catch (error) {
280 console.error('Error accepting invitation:', error);
281 return { success: false, message: 'Failed to accept invitation' };
282 }
