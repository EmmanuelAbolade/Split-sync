23export const trackConnection = async (userId, otherUserId, groupId, groupName) => {
24 if (userId === otherUserId) return; // Don't connect user to themselves
25
26 try {
27 // Get user info
28 const otherUserDoc = await getDoc(doc(db, 'users', otherUserId));
29 if (!otherUserDoc.exists()) return;
30
31 const otherUserData = otherUserDoc.data();
32
33 // Connection data
34 const connectionData = {
35 userId: otherUserId,
36 name: otherUserData.fullName || otherUserData.displayName || 'Unknown',
37 email: otherUserData.email || '',
38 photoURL: otherUserData.photoURL || null,
39 lastGroupTogether: groupName,
40 lastGroupId: groupId,
41 updatedAt: serverTimestamp()
42 };
43
44 // Check if connection exists
45 const connectionRef = doc(db, 'connections', userId, 'connectedUsers', otherUserId);
46 const existingConnection = await getDoc(connectionRef);
47
48 if (existingConnection.exists()) {
49 // Update existing connection
50 await updateDoc(connectionRef, {
51 sharedGroups: arrayUnion({
52 groupId,
53 groupName,
54 addedAt: new Date()
55 }),
56 lastGroupTogether: groupName,
57 lastGroupId: groupId,
58 updatedAt: serverTimestamp()
59 });
60 } else {
61 // Create new connection
62 await setDoc(connectionRef, {
63 ...connectionData,
64 sharedGroups: [{
65 groupId,
66 groupName,
67 addedAt: new Date()
68 }],
69 connectedAt: serverTimestamp()
70 });
71 }
72
73 console.log(`✅ Connection tracked: ${userId} <-> ${otherUserId} in ${groupName}`);
74 } catch (error) {
75 console.error('Error tracking connection:', error);
76 }
77};

104export const getConnections = async (userId) => {
105 try {
106 const connectionsRef = collection(db, 'connections', userId, 'connectedUsers');
107 const snapshot = await getDocs(connectionsRef);
108
109 // ⭐ Fetch FRESH user data from users collection
110 const connections = await Promise.all(
111 snapshot.docs.map(async (docSnap) => {
112 const connectionData = docSnap.data();
113 const connectedUserId = docSnap.id;
114
115 try {
116 // Fetch current user data
117 const userDoc = await getDoc(doc(db, 'users', connectedUserId));
118
119 if (userDoc.exists()) {
120 const userData = userDoc.data();
121 // Use CURRENT name from users collection
122 return {
123 id: connectedUserId,
124 userId: connectedUserId,
125 name: userData.displayName || userData.fullName || connectionData.name || 'Unknown',
126 email: userData.email || connectionData.email || '',
127 photoURL: userData.photoURL || connectionData.photoURL || null,
128 // Keep connection metadata
129 sharedGroups: connectionData.sharedGroups,
130 lastGroupTogether: connectionData.lastGroupTogether,
131 lastGroupId: connectionData.lastGroupId,
132 updatedAt: connectionData.updatedAt,
133 connectedAt: connectionData.connectedAt,
134 };
135 }
136 } catch (err) {
137 console.log(`Could not fetch fresh data for ${connectedUserId}, using cached`);
138 }
139
140 // Fallback to cached data if user fetch fails
141 return {
142 id: connectedUserId,
143 userId: connectedUserId,
144 ...connectionData
145 };
146 })
147 );
148
149 // Sort by most recent connection
150 connections.sort((a, b) => {
151 const aTime = a.updatedAt?.toMillis() || 0;
152 const bTime = b.updatedAt?.toMillis() || 0;
153 return bTime - aTime;
154 });
155
156 return connections;
157 } catch (error) {
158 console.error('Error getting connections:', error);
159 return [];
160 }

164export const syncMyConnectionsFromGroups = async (userId) => {
165 try {
166 if (!userId) return { success: false, synced: 0 };
167
168 // Get all groups where I'm a member
169 const groupsQuery = query(
170 collection(db, 'groups'),
171 where('memberIds', 'array-contains', userId)
172 );
173
174 const snap = await getDocs(groupsQuery);
175
176 let synced = 0;
177
178 for (const groupDoc of snap.docs) {
179 const groupData = groupDoc.data();
180 const groupName = groupData?.name || 'Group';
181 const memberIds = groupData?.memberIds || [];
182
183 // For each other member in the group, ensure I have a connection doc
184 for (const otherUserId of memberIds) {
185 if (!otherUserId || otherUserId === userId) continue;
186
187 await trackConnection(userId, otherUserId, groupDoc.id, groupName);
188 synced++;
189 }
190 }
191
192 return { success: true, synced };
193 } catch (error) {
194 console.error('Error syncing my connections from groups:', error);
195 return { success: false, synced: 0, error: error.message };
196 }

285export const getSuggestedConnections = async (userId) => {
286 try {
287 // Get user's direct connections
288 const myConnections = await getConnections(userId);
289 const myConnectionIds = myConnections.map(c => c.userId);
290
291 // Get connections of connections (2nd degree)
292 const suggestedMap = new Map();
293
294 for (const connection of myConnections) {
295 const theirConnections = await getConnections(connection.userId);
296
297 for (const theirConnection of theirConnections) {
298 // Skip if it's the user themselves or already connected
299 if (theirConnection.userId === userId || myConnectionIds.includes(theirConnection.userId)) {
300 continue;
301 }
302
303 // Count mutual connections
304 if (suggestedMap.has(theirConnection.userId)) {
305 const existing = suggestedMap.get(theirConnection.userId);
306 existing.mutualConnections++;
307 existing.mutualWith.push(connection.name);
308 } else {
309 suggestedMap.set(theirConnection.userId, {
310 ...theirConnection,
311 mutualConnections: 1,
312 mutualWith: [connection.name]
313 });
314 }
315 }
316 }
317
318 // Convert to array and sort by mutual connections
319 const suggested = Array.from(suggestedMap.values());
320 suggested.sort((a, b) => b.mutualConnections - a.mutualConnections);
321
322 return suggested.slice(0, 10); // Top 10 suggestions
323 } catch (error) {
324 console.error('Error getting suggested connections:', error);
325 return [];
326 }
